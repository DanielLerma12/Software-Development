
--install eslint extension on IDE:

npm add eslint -D
npm add @eslint/js -D // Js config
npm add globals -D // list of global variables

----------------------------------------------

--configure for js:

-eslint.config.js file for node variables:

import js from "@eslint/js";
import globals from "globals";

export default [
  js.configs.recommended,
  {
    languageOptions: {
      globals: {
        ...globals.node
      }
    }
  }
];

-eslint.config.js file for browser variables:

import globals from "globals";

export default [
  {
    languageOptions: {
      globals: {
        ...globals.browser,
      },
    },
  },
];

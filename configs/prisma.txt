
--Prisma general config:

-install:

npm install prisma @prisma/client

-initialize:

npx prisma init

-configure schema.prisma, for example:

generator client {
  provider = "prisma-client"
  output   = "../generated/prisma"
}

datasource db {
  provider = "postgresql"
}

-configure client:

npx prisma generate // prisma client for reading the schema, using the database and queries, run it each time the schema changes

-client:

lib/prisma.js // folder

import "dotenv/config";
import { PrismaClient } from "./generated/prisma/client.ts";
import { PrismaPg } from "@prisma/adapter-pg";

const adapter = new PrismaPg({
  connectionString: process.env.DATABASE_URL,
});

export const prisma = new PrismaClient({
  adapter,
}); // dotenv, prisma client and adapters

-usage example:

import { prisma } from "../../lib/prisma.js";

const movies = await prisma.movies.findMany();

--usage with postgresql:

-install:

npm install pg
npm install @prisma/adapter-pg

-configure .env:

DATABASE_URL="postgresql://postgres:password@db.xxxxx.supabase.co:5432/postgres"

-seed, inside package.json:

  "prisma": {
    "seed": "node prisma/seed.js"
  },

-push and pull:

npx prisma db push // create the database after creating the schema
npx prisma db pull // if the database is already created, do this for each database update, then run npx prisma generate

-npx prisma generate for production build command
